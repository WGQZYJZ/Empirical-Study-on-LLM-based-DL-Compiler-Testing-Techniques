
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv  = torch.nn.Conv2d(3,8,1)
 
    def forward(self, x1):
        v1  = self.conv(x1)
        v2  = v1  * 0.5 #v1 = 0.75641595 * 1.4553654
        v3  = v1 ** 2 #v1 = -0.77805102 * -0.5, v3 = (v1)^(2) = (-0.39927953)^(2) = 0.1596499
        v4  = v3 ** v1 #v1 = 0.8142092, v4 = v3^(v1) = ((v1)^(2))^(v1), v4 = (0.1596499)^(0.8142092), v4 = 0.27581
        v5  = v4 * -0.044715 #v1 = 0.380173, v4 = 0.36109345 * (-0.044715), v5 = 0.280909
        v6  = v1 + v5 #v1 = -0.01550152, v5 = 0.280909, v6 = (-0.39927953)^(2) + (v1)^(-0.044715), v6 = 0.23377
        v7  = v6 * -0.79788 #v1 = 0.570454, v7 = 0.23377 * (-0.79788) = ((-0.39927953)^(2))^(v1), v6 = 0.13592
        v8  = torch.tanh(-0.442002 + (torch.abs(x1))) #v1 = -0.399279, v7 = (-0.79788) * ((-0.3992795)^(v1)), v6 = 0.13592, 0.442002 - (abs(x1)) = torch.abs((-0.39927953)^(v1)), 0.442002 - (-0.8) = v8
        v9  = v8 + 1 #v1 = -0.7, v7 = -0.6514205 * ((-0.399279)^(v1)) - (torch.abs(x1)), v7 = (-0.39927953)^(2), 1.458 + torch.abs(-0.399279) = v9
        v10 = v2 * v9 #v1 = -0.6514205, v2 = 0.5, v9 = (-0.6514205) * ((-0.3992795)^(v1)) - (torch.abs(-0.399279)), torch.abs((x1)) = abs((-0.39927953)^(v1)), v10 = 0.48617
        return v10
