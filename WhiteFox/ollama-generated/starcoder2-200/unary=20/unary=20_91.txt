
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.convt  = torch.nn.ConvTranspose2d(3, 8, 1)
 
    def forward(self, x1):
        v1  = self.convt(x1) # applying pointwise transposed convolution to the input tensor
        v2  = torch.sigmoid(v1) 
        return v2


# Initializing the model
m  = Model() 

# Inputs to the model
x1 = torch.randn(1, 3, 64, 64)
__output__  = m(x1)
