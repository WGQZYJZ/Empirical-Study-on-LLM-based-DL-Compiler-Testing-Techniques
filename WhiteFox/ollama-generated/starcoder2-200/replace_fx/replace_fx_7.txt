import torch

class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()

    def forward(self, input1): 
        output = torch.nn.functional.dropout(input1) # Apply dropout to the input tensor
        output = torch.rand_like(output)              # Generate a random tensor with the same shape and type as `output`
        return output

# Initializing the model
m  = Model()


# Inputs to the model
x1 = torch.randn(3,4,5) 
__output__  = m(x1)
