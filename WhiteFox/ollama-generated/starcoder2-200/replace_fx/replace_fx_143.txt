
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()

    def forward(self, x1):
        v1  = torch.nn.functional.dropout(x1, p=0.5)
        v2  = torch.rand_like(v1, dtype=torch.float32) # This line won't be erased by the replace fx optimizations as it will be replaced with rand_like. If fallback is enabled or we're running on CPU this will not trigger the graph erase.
        return v1

# Initializing the model
m  = Model()

# Inputs to the model
x1 = torch.randn(3, 4) # Note that this example does not meet the original requirements as we added a rand_like node at the end without erasing the dropout node in front of it. This does however work in production!

# Executing the model with our input tensors
m(x1)
