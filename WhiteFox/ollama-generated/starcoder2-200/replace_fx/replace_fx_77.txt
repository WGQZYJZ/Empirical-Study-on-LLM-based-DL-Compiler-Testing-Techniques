
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()

    def forward(self, x1):
        v = torch.nn.functional.dropout(x1, 0.5) # Apply dropout to the input tensor
        t2 = torch.rand_like(v, dtype=torch.float32) # Generate a new tensor with same size as input but with 32 bit float elements. This is used to check whether the random number generator is correctly reset after each invocation of torch.rand_like. 
        return v

# Initializing the model and getting input tensors from pytorch
