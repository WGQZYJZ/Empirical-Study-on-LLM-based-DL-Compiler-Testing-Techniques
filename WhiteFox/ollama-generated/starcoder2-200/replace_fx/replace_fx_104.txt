
```
import torch
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear = torch.nn.Linear(2, 2)

    def forward(self, x1): 
        v1 = torch.nn.functional.dropout(x1, 0.3) 
        v2 = torch.rand_like(v1, dtype=torch.float64)
        return self.linear(v2).clone()
m = Model()
```
