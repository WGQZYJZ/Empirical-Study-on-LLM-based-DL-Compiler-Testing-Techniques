
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
 
    def forward(self, x1):
        v1 = torch.relu(x1)  # Apply the Rectified Linear Unit function to input tensor
        v2 = torch.clamp_min(v1, min=0.)  # Clamp the output of the previous operation to a minimum value equal to 0.
        return torch.clamp_max(v2, max=32768)


# Initializing the model
m = Model()
 
# Inputs to the model
x1 = torch.randn(1, 3, 4096, 503)
__output__  = m(x1)
