
class Model(torch.nn.Module):
    def __init__(self, minv=0., maxv=255):
        super().__init__()
        self.conv  = torch.nn.Conv2d(3,8,1)
        self._minv  = torch.nn.Parameter(torch.Tensor([float(minv)]))
        self._maxv  = torch.nn.Parameter(torch.Tensor([float(maxv)]))
 
    def forward(self, x):
        v1  = self.conv(x)
        v2  = torch.clamp_min(v1, self._minv).clamp_max(v1, self._maxv)
