
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()

    def forward(self, x1):
        v1 = torch.nn.functional.linear(x1) # Apply linear transformation to the input tensor.
        v2  = v1.permute(0, 2, 1)            # Permute the output tensor of the linear function.
        return v2

# Initializing the model
m  = Model()

# Inputs to the model. x1 has more than two dimensions. It will be permuted at the final step after the linear operation is invoked on it.
x1 = torch.randn(5, 3, 4)
__output__= m(x1).permute([0,2,1])
