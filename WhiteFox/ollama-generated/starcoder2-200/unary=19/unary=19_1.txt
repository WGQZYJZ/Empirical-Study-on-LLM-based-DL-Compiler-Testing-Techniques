
```
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear = torch.nn.Linear(32 * 32 * 3, 10)
 
    def forward(self, x1):
        v1 = self.linear(x1)
        v2 = torch.sigmoid(v1)
        return v2
```
# Initializing the model
m = Model()

 # Inputs to the model
__input_tensor__  = torch.randn(3200, 32 * 32 * 3)
