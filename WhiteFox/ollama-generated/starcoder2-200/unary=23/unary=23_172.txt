        self.convtranspose  = torch.nn.ConvTranspose2d(3, 8, 1, stride=1)
 
    def forward(self, x1):

        v2 = torch.tanh(v1) # Apply the hyperbolic tangent function to the output of the transposed convolution
        return v2
