
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()

    def forward(self, x1):
        v1  = torch.split(x1, [32], dim=0)
        return torch.cat([v1[i] for i in range(len(v1))], dim=0).sum()


# Initializing the model
m = Model()

# Inputs to the model
x1  = torch.randn(64, 32, 32)
__output__  = m(x1)
```
