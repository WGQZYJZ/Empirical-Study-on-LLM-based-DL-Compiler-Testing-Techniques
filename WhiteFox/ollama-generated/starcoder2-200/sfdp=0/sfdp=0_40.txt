```
class ScaledDotProductAttention(torch.nn.Module):
    def __init__(self, scale=None):
        super().__init__()
        self.scale = torch.tensor(1., dtype=torch.float) if scale is None else scale
 
    def forward(self, query, key, value):
        scaled_dot_product  = torch.matmul(query, key.transpose(-2, -1)) / self.scale 
        attention_weights  = scaled_dot_product.softmax(dim=-1) # (..., N, Q, K)
        output  = attention_weights.matmul(value)              # (..., N, Q, V)
 
        return output
```
