    def __init__(self):
        super().__init__()
        self.conv = torch.nn.ConvTranspose2d(8, 3, kernel_size=1, stride=1, padding=0)

    def forward(self, x1):
         v1 = self.conv(x1)
         v2 = v1 * 0.5
         v3 = (v1 * v1) * v1
         v4 = torch.sum(v3, dim=-1).permute(-1,)
         v6 = v4 + x1
         v7 = torch.tanh((x1 + v6))
