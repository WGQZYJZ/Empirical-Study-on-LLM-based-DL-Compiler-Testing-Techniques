
```python
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv = torch.nn.Conv2d(3, 8, 1, stride=1, padding=1)
 
    def forward(self, x1):
        v1 = self.conv(x1) # Applies a pointwise convolution with kernel size 1 to the input tensor.
        v2 = torch.sigmoid(v1) # Applies sigmoid activation function to the output of the convolution.
        return v2
```


# Initializing the model
```python
m = Model()
```
