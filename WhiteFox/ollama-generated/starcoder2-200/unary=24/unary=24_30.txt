
class Model(torch.nn.Module):
    def __init__(self, negative_slope=0.1):
        super().__init__()
        self.conv = torch.nn.Conv2d(3, 8, 1) # Apply pointwise convolution with kernel size 1 to the input tensor
        self.negative_slope = negative_slope

    def forward(self, x1):
        v1  = self.conv(x1)
        v2 = (v1 > 0).to(torch.float32) 
        v3 = v1 * -self.negative_slope # Multiply the output of the convolution by the negative slope
        v4 = torch.where(v2, v1, v3)# Apply the where function to select elements from t1 or t3 based on the mask t2
        return v4


# Initializing and testing the model
m  = Model()
x1 = torch.randn(1, 3, 64, 64)
__output__  = m(x1)
