
class Model(torch.nn.Module):
    def __init__(self, query, key, scale_factor, dropout_p):
        super().__init__()
        self.query  = torch.nn.Parameter(query)
        self.key  = torch.nn.Parameter(key)
        self.scale_factor  = scale_factor
        self.dropout  = torch.nn.Dropout(dropout_p)
 
    def forward(self, value):
        qk  = torch.matmul(self.query, self.key.transpose(-2, -1)) 
        scaled_qk  = qk.mul(self.scale_factor) 
        softmax_qk  = scaled_qk.softmax(dim=-1) 
        dropout_qk  = self.dropout(softmax_qk)
        output  = dropout_qk @ value 
        return output

# Initializing the model
query, key, scale_factor, dropout_p  = torch.randn(32, 512), torch.randn(32, 512), 0.9782643844604492, 0.5310903787612915
m = Model(query=query, key=key, scale_factor=scale_factor, dropout_p=dropout_p)
 
# Input tensors for the model
x1  = torch.randn(4, 512) # This tensor contains the value to compute its dot product with each row of a query and a key tensor. It will be broadcasted to shape (32, 512).
x2  = m.query @ m.key.transpose(-2, -1) * scale_factor
x3  = x2.softmax(dim=-1)
x4  = torch.nn.functional.dropout(x3, p=dropout_p)
__output__  = x4 @ value # We multiply the dropout output by each row of a value tensor. We will broadcast this tensor to shape (4, 512).
