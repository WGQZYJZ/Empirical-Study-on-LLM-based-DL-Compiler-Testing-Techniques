```
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.convt = torch.nn.ConvTranspose2d(3, 8, 1, stride=1, padding=0)
 
    def forward(self, x1):
        v1  = self.convt(x1)
        v2  = torch.sigmoid(v1) 
        return v2 * v1
```
# Initializing the model
m = Model()

 # Inputs to the model
input_tensor = torch.randn(3, 64, 64, 64)
__output__   = m(input_tensor)
