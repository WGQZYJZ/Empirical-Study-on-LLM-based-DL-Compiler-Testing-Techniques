```
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
 
        self.conv1 = torch.nn.Conv2d(3, 8, 1)
 
    def forward(self, x1):
        v0 = self.conv1(x1)
        v2 = torch.sigmoid(v0) 
        v4 = v0 * v2
        return v4
```

# Initializing the model
m = Model()

# Inputs to the model
x1  = torch.randn(1, 3, 64, 64)
