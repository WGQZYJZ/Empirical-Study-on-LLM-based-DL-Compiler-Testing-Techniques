class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()

    def forward(self, query, key, value):
        v1  = torch.matmul(query, key.transpose(-2, -1)) # Compute the dot product of the query and the key 
        v2  = v1 / 0.875                             # Scale the dot product by the inverse scale factor (0.875)
        v3  = v2.softmax(dim=-1)                       # Apply softmax to the scaled dot product
        v4  = torch.nn.functional.dropout(v3, p=0.1, training=self.training)   # Apply dropout to the softmax output with a dropout probability of 0.1 (p = 0.9 is set by default in PyTorch)
        v5  = v4.matmul(value)                         # Compute the dot product of the dropout output and the value
        return v5
