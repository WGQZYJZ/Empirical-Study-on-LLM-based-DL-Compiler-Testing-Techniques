
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
 
    def forward(self, x1, y1):
        v2 = torch.mm(x1, y1)
        v3 = torch.cat([v2 for _ in range(9)], 0)
        return v3


m = Model()

 # Initializing the model
m.eval()
 
# Inputs to the model
x1  = torch.randn(3, 8)
y1  = torch.randn(8, 5)
 
 __output__= m(x1, y1)
