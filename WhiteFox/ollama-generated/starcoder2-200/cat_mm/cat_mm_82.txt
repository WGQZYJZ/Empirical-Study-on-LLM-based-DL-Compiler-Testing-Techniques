
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
 
    def forward(self, x1, x2):
        v0  = torch.randn(35)
        v0  = torch.cat([v0] * len(x1)) 
        return (v0, v0), ((torch.mm(x1, x2)), v0, )
 
# Initializing the model
m  = Model()

 # Inputs to the model
x1 = [None] + list(map(lambda _: torch.randn((35,47)), range(98)))
x2 = [None] + list(map(lambda _: torch.randn((47,100)), range(96)))
__output__  = m(x1[1], x2[-1])
