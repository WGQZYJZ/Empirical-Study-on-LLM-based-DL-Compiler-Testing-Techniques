
```
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear = torch.nn.Linear(320,1)
 
    def forward(self, x1):
        v1 = self.linear(x1) + 0  # Add another tensor to the output of the linear transformation
        return v1
```

# Initializing the model
m = Model()


# Inputs to the model
x1 = torch.randn(16,320)
__output__  = m(x1)
