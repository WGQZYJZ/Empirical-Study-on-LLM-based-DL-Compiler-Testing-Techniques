
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
 
    def forward(self, x1, other):
        v1 = torch.nn.functional.linear(x1)  # Applying a linear transformation to the input tensor
        v2 = v1 + other
        return v2


# Initializing the model
m = Model()


# Inputs to the model
x1  = torch.randn(3,4)
other  = torch.randn(4,)


__output__  = m(x1, other)
