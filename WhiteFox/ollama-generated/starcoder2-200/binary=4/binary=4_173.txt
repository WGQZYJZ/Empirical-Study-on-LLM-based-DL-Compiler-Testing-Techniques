
```
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear = torch.nn.Linear(200, 5)
 
    def forward(self, x1):
        v1 = self.linear(x1)
        return v1 + torch.randn(v1.size())


m = Model()
```

# Initializing the model
m = Model()

# Inputs to the model
x1  = torch.randn(5, 200)

# Expected output of the model
__output__  = m(x1).detach().cpu().numpy()
