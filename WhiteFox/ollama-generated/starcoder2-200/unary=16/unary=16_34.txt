
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
 
        self.linear = torch.nn.Linear(256, 4)
 
    def forward(self, x1):
        v0 = x1 + 2
        
        # 1
        v1 = self.linear(v0)
        # 3
        v3 = torch.relu(v1)
        
        return v3

# Initializing the model
m = Model()


# Inputs to the model
x1 = torch.randn(5, 256)
__output__  = m(x1)
