class Model(torch.nn.Module):
    def __init__(self, minv=None, maxv=None):
        super().__init__()
        self.lin = torch.nn.Linear(2048 * 16, 32)
 
        if minv is not None:
            self.clamp_min = torch.nn.Clamp(min=minv)
        else:
            self.clamp_min = None
 
        if maxv is not None:
            self.clamp_max = torch.nn.Clamp(max=maxv)
        else:
            self.clamp_max = None
 
    def forward(self, x1):
        v1  = self.lin(x1)
 
        if self.clamp_min is not None and self.clamp_max is not None:
            v2 = torch.clamp(v1, min=self.clamp_min, max=self.clamp_max) 
        elif self.clamp_min is not None or self.clamp_max is not None: 
            if self.clamp_min is not None and self.clamp_max is None: 
                v2 = torch.clamp(v1, min=self.clamp_min)
            else:
                v2 = torch.clamp(v1, max=self.clamp_max)
        else:
            raise Exception('There should be at least one clamp value')
 
        return v2


# Initializing the model with a minimum and maximum value
m  = Model()


# Inputs to the model
x1 = torch.randn(307, 512 * 4)
 
# Running inference on the model (note that `clamp_min` and `clamp_max` values are randomly generated by the user during the execution of this system and may vary between each run)
__output__  = m(x1)
