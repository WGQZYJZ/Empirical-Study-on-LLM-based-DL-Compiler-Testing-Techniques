
class Model(torch.nn.Module):
    def __init__(self, max_value=500, min_value=-500):
        super().__init__()
        self.linear  = torch.nn.Linear(32*64*64, 8)
 
    def forward(self, x1):
        v1  = self.linear(x1.view(-1)) 
        v2  = torch.clamp_min(v1, min_value=self._min_value)
        v3  = torch.clamp_max(v2, max_value=self._max_value)
        return v3


# Initializing the model with custom minimum and maximum values
m  = Model()


# Inputs to the model
x1  = torch.randn(64, 8*32*32)
__output__  = m(x1)

# Description of requirements
The model should contain the following pattern:
```
t1 = Conv2d(256, 728, kernel_size=(7, 7), stride=(2, 2))
t3 = t1 * 0.9 # Multiply the output by a constant value of 0.9
t4 = torch.tanh(t3) + 0.3 # Add to the previous output a constant value of 0.3
```
