```
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
 
    def forward(self, inp1, inp2):
        v0 = torch.randn(3)
        v1  = torch.mm(inp1, inp2) # Matrix multiplication on two input tensors 
        v2  = v1 + v0 # Add the result of the matrix multiplication to another tensor 'v0'
 
        return v2
```
