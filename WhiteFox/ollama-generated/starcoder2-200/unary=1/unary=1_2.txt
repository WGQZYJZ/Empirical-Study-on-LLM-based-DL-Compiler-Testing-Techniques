
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear  = torch.nn.Linear(1024, 512)
 
    def forward(self, x1):
        v1  = self.linear(x1)
        v2  = v1 * 0.5
        v3  = (v1 + v1*v1*v1)*0.044715 
        v4  = torch.tanh(v3*0.7978845608028654) # tanh
        v5  = v4+1.0                          # Add 1 to hyperbolic tangent 
        v6  = v2 * v5                         # Multiplication of the linear transformation and hyperbolic tangent 
        return v6
