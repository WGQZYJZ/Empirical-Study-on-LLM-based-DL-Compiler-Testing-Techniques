# Initialize a Conv2d module without bias
conv = torch.nn.Conv2d(3, 4, kernel_size=5)

# Initialize a BatchNorm2d module with affine=False
bn1  = torch.nn.BatchNorm2d(4, affine=True)
bn2  = torch.nn.BatchNorm2d(4, affine=False)

class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()

    def forward(self, x1):
        conv_out  = self._conv(x1)

        # Apply batch normalization in evaluation mode without tracking running statistics 
        bn2_out   = self._bn2(conv_out)

        # Apply batch normalization with affine=True.
        bn1_out   = self._bn1(conv_out)
        return conv_out, bn1_out, bn2_out
```

# Initializing the model
m  = Model()


# Inputs to the model
x1 = torch.randn(3, 4, 60, 57)
__output__  = m(x1)
