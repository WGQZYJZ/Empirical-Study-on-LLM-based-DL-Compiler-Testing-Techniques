
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv  = torch.nn.Conv2d(3, 8, 1)
 
    def forward(self, x1):
        v1 = self.conv(x1)
        v6 = v1 + torch.randn(v1.shape[0], v1.shape[1]) # Add another tensor to the output of the convolution
        return v6


# Initializing the model
m  = Model()
 
# Inputs to the model
x1 = torch.randn(1, 3, 8, 8)
__output__  = m(x1)
