
class Model(torch.nn.Module):
    def __init__(self, negative_slope=0.25):
        super().__init__()
        self.conv  = torch.nn.ConvTranspose2d(3, 8, 1, stride=1, padding=0)
 
    def forward(self, x1): 
        v1  = self.conv(x1) # Pointwise transposed convolution is applied to the input tensor.
        msk = (v1 > 0).type_as(v1)# Create a mask where each element is True if the corresponding element in v1 is greater than 0, False otherwise.
        slr = -negative_slope * torch.ones_like(v1) # A negative slope is used to multiply the elements of v1
        otp = msk * v1 + (1-msk)*slr # Apply the where function to select elements from v1 or the result of the multiplication based on the mask.
        return otp


# Initializing the model and defining the parameters
negative_slope  = torch.tensor(0.25)
m  = Model(negative_slope=negative_slope) 

 # Inputs to the model
x1  = torch.randn(1, 3, 64, 64)

 __output__  = m(x1)
