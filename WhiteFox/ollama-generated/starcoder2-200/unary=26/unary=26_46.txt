```
class Model(torch.nn.Module):
    def __init__(self, negative_slope=0.5):
        super().__init__()
        self.conv = torch.nn.ConvTranspose2d(32, 16, 4)
 
    def forward(self, x1):
        v1 = self.conv(x1)
        mask = v1 > 0
        slope  = negative_slope
        v2  = torch.where(mask, v1, -slope * v1) 
        return v2
```

# Initializing the model
m  = Model()

# Inputs to the model
x1 = torch.randn(32, 16, 50, 70)
