
class Model(torch.nn.Module):
    def __init__(self, negative_slope=0.15):
        super().__init__()
        self.conv  = torch.nn.ConvTranspose2d(3, 8, kernel_size=(4, 4), stride=1)

    def forward(self, x1):
        v1 = self.conv(x1)
        mask = v1 > 0
        v2 = negative_slope * v1
        v3 = torch.where(mask, v1, v2) # replace the output of the convolution with -negative_slope * v1 where mask is True
        return v3


# Initializing the model
m = Model()


# Inputs to the model 
x1 = torch.randn(10, 8, 40, 50) # shape: [10 x 8 x 40 x 50]
__output__  = m(x1)
