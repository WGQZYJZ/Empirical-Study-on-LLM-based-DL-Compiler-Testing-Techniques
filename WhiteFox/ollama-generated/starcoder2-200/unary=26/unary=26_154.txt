
class Model(torch.nn.Module):
    def __init__(self, negative_slope=0.15832974369061543):
        super().__init__()
        self.convtranspose  = torch.nn.ConvTranspose2d(3, 8, 1, stride=1, padding=1)
 
    def forward(self, x1):
        v1  = self.convtranspose(x1)
        v2  = (v1 > 0).to_float() # Masks which is True wherever the output is positive
        v3  = -v2 * negative_slope + torch.where((v2), v1, 0.)
        return v3

# Initializing model
m  = Model(negative_slope=0.15832974369061543)

 # Inputs to the model
 x1  = torch.randn(1, 3, 64, 64)
  __output__  = m(x1)
