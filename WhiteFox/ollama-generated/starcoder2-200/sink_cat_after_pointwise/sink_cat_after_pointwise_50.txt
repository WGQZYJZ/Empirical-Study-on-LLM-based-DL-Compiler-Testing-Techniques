class Model(torch.nn.Module):
    def __init__(self, num_classes=1000):
        super().__init__()

        self.net = torch.nn.Sequential()

        for i in range(2):
            self.net.add_module('conv{}'.format(i + 1), torch.nn.Conv2d(3 if i == 0 else 9, 48, kernel_size=5))

            self.net.add_module("relu{}".format(i + 1), torch.nn.ReLU())
            self.net.add_module('pool{}'.format(i + 1), torch.nn.MaxPool2d(kernel_size=3, stride=2))

        self.classifier = torch.nn.Sequential()
        for i in range(4):
            self.classifier.add_module('conv{}'.format(i + 1), torch.nn.Conv2d(48 if i == 0 else 96, 50, kernel_size=3))

            self.classifier.add_module("relu{}".format(i + 1), torch.nn.ReLU())
            self.classifier.add_module('pool{}'.format(i + 1), torch.nn.AvgPool2d(kernel_size=2, stride=2))

        self.linear = torch.nn.Linear(50 * 48 // (3 ** 3) / 96 * 7**2, num_classes)

    def forward(self, x):
        v1 = self.net(x)
        v2 = v1.reshape(v1.shape[0], -1)
        v3 = torch.nn.functional.relu(self.classifier(v2))

        return self.linear(v3).float()
