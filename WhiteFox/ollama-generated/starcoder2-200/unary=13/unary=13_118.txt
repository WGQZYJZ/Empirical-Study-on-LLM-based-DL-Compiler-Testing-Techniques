
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.lin  = torch.nn.Linear(1024, 512)
    
    def forward(self, x1):
        v1  = self.lin(x1) # Apply a linear transformation to the input tensor
        v2  = torch.sigmoid(v1) 
        v3  = v1 * v2   # Multiply the output of the linear transformation by the output of the sigmoid function
