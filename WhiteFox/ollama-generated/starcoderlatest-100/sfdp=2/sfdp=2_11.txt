
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
 
        # 1x1 convolution to convert the query tensor to 2-d representation, since it will be transposed
        self.query_proj = torch.nn.Conv2d(3, 8, kernel_size=1) 
        # 1x1 convolution to convert the key tensor to 2-d representation, since it will be transposed
        self.key_proj = torch.nn.Conv2d(3, 8, kernel_size=1)
 
    def forward(self, query, key):
        v6 = self.attention(query, key)
        return v6
 
    def attention(self, query, key):
        # Transpose the tensor before performing the dot product for efficiency
        query = query.permute(0, 2, 3, 1)
        key = key.permute(0, 2, 3, 1)
 
        # Concatenate the queries and keys
        v1 = torch.cat((query, key), dim=1) 
        
        # Project queries and keys to a 2-d representation for better computations
        qk = self.query_proj(v1)
        kq = self.key_proj(v1)
 
        # Compute the dot product of the query and key
        v6 = torch.matmul(qk, kq.transpose(-2, -1))
        return v6
