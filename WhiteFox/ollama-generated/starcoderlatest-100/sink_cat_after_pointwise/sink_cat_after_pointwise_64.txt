
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear = torch.nn.Linear(2, 2)

    def forward(self, x1, x2):
        v1 = torch.cat([x1, x2], dim=0) # Concatenate inputs along the dimension with index 0
        v2 = v1.view(v1.shape[0], -1) # Reshape the concatenated tensor (i.e., reshape it into a vector)

        v3 = torch.relu(v2)
        return v3
