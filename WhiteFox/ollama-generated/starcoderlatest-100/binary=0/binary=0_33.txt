
class Model(torch.nn.Module):
    def __init__(self, other=None):
        super().__init__()
        self.conv = torch.nn.Conv2d(3, 8, 1, stride=1, padding=1)
        if other is None:
            raise ValueError('The value of `other` must be not `None`.')
        else:
            self.other = other
 
    def forward(self, x1):
        v1 = self.conv(x1)
        __output__  = v1 + self.other
        return __output__
```
