
class Model(torch.nn.Module):
    def __init__(self, other_tensor=None):
        super().__init__()
        if other_tensor:
            self.conv = torch.nn.Conv2d(3, 8, 1, stride=1, padding=1)
        else:
            self.conv = torch.nn.Conv2d(3, 8, 1, stride=1, padding=1)
 
    def forward(self, x1):
        if self.conv:
            v1 = self.conv(x1) + other_tensor

            v1 = self.conv(x1)
