
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear = torch.nn.Linear(2, 2)

    def forward(self, x1):
        v1 = torch.nn.functional.dropout(x1, p=0.4, training=True)
        #v1 = torch.rand_like(x1)
        return self.linear(v1)


# Initializing the model with a random input and output
m = Model()
input = torch.randn(200, 6, 3)
output = m(input)
torch._C._log_api_usage_once("aten", "rand_like") # This line triggers rand_like op in the graph of m
