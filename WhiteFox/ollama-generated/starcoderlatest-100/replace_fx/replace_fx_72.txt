
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()

    def forward(self, x1):
        v1 = torch.rand_like(x1, dtype=torch.float64) # Replace random operation with corresponding replacement node in the graph
        v2 = torch.nn.functional.dropout(v1, p=0.5) # Dropout the tensor generated by the `torch.rand_like` function
        return v2


# Initializing the model
m = Model()


# Inputs to the model
x1 = torch.randn(1, 2, 2)
