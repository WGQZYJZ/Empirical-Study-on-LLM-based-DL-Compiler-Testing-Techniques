
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.lin = torch.nn.Linear(32 * 64 * 64, 32)
 
    def forward(self, x1):
        v1 = self.lin(x1.view(-1, 32 * 64 * 64))
        v2 = torch.sigmoid(v1)
        return v2
