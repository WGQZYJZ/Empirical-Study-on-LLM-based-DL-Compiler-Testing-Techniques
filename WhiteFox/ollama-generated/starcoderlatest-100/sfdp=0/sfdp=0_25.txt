
class ScaledDotProductAttention(torch.nn.Module):
    def __init__(self):
        super().__init__()
 
    def forward(self, query, key, value, inv_scale):
        scaled_dot_product = torch.matmul(query, key.transpose(-2, -1)) / inv_scale
        attention_weights = scaled_dot_product.softmax(dim=-1)
        output = attention_weights.matmul(value)
        return output
 

class MultiHeadAttention(torch.nn.Module):
    def __init__(self):
        super().__init__()
 
    def forward(self, query, key, value):
        dim_head  = query.shape[-1] // self.num_heads  # dimension of attention head
        
        scaled_dot_product_attention = ScaledDotProductAttention()
 
        linear_out_q = torch.nn.Linear(query.shape[0], query.shape[1]*self.num_heads)
        linear_out_k = torch.nn.Linear(key.shape[0], key.shape[1]*self.num_heads)
        linear_out_v = torch.nn.Linear(value.shape[0], value.shape[1]*self.num_heads)
 
        scaled_dot_product_attention_out = scaled_dot_product_attention(
            linear_out_q(query), linear_out_k(key), linear_out_v(value), 
            inv_scale=dim_head ** 0.5
        )
 
        return scaled_dot_product_attention_out


class PositionalEncoding(torch.nn.Module):
    def __init__(self, d_model, dropout, max_len=5000):
        super().__init__()
 
        pos = torch.arange(max_len).unsqueeze(1)  # create a (1 x sequence length) tensor of indices for later use
        emb = torch.sin(pos / (10000 ** ((2 * dim_model) // max_len)))  # calculate sinusoids for later use
 
        # the first dimension is added to the embedding because it's used in addition to position encoding with this line:
        emb = emb.unsqueeze(0)  # add a batch axis (batch size = 1 -> batch x sequence length)
 
        # create embeddings matrix as (dim_model, seq_length) using the created sinusoids tensor and transpose it so that it has shape (seq_length, dim_model):
        self.emb = emb[:, :d_model]
 
    def forward(self, input):
        return self.emb * torch.transpose(input, -2, -1)


class EncoderLayer(torch.nn.Module):
    # TODO: use nn.TransformerEncoderLayer for layers that require multi-head attention (e.g., positionwise and feedforward neural networks)
 
    def __init__(self, dim_model, dff, num_heads, dropout):
        super().__init__()
 
        self.att = MultiHeadAttention()
 
        # TODO: add PositionalEncoding if specified in input (e.g., in TransformerEncoderLayer to be used for encoding source sequences)
        self.pos_emb = PositionalEncoding(d_model=dim_model)
 
        self.ffn = torch.nn.Sequential(
            torch.nn.Linear(dff, dff),
            torch.nn.ReLU(),
            torch.nn.Dropout(dropout),
            torch.nn.Linear(dff, dim_model),
        )
 
    def forward(self, x):
        return self.ffn(self.pos_emb(x) + self.att(x, x, x))
 

class TransformerEncoder(torch.nn.Module):
    # TODO: use nn.TransformerEncoder for layers that require multi-head attention (e.g., positionwise and feedforward neural networks)
 
    def __init__(self, transformer_encoder_layers, dropout=0.1):
        super().__init__()
 
        self.transformer_encoder_layers = torch.nn.ModuleList(transformer_encoder_layers)
 
        self.dropout = torch.nn.Dropout(dropout)
 
    def forward(self, x):
        return torch.stack([layer(x) for layer in self.transformer_encoder_layers], dim=0)  # concatenate the outputs of all layers


class TransformerModel(torch.nn.Module):
    def __init__(self, transformer_encoder_layers, output_dim):
        super().__init__()
 
        # TODO: use nn.TransformerEncoder for layers that require multi-head attention (e.g., positionwise and feedforward neural networks)
        self.transformer_encoder = TransformerEncoder(transformer_encoder_layers, dropout=0.1)
 
        self.dropout = torch.nn.Dropout(dropout)
        self.output = torch.nn.Linear(d_model, output_dim)
 
    def forward(self, x):
        return self.output(self.dropout(self.transformer_encoder(x)))
