
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.layer_norm1 = LayerNorm(4)
        self.conv  = torch.nn.Conv2d(3, 8, 1, stride=1, padding=1)
 
    def forward(self, x1):
        v1  = self.layer_norm1(x1)
        v2  = self.conv(v1)
        return v2
