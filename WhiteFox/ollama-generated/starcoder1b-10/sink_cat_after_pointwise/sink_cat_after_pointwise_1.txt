
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear = torch.nn.Linear(2, 2)

    def forward(self, x1):
        t1  = torch.cat([x1, ...], dim=0).view(-1, 2)
        t2 = torch.relu(t1)
        return t2
