
class Attention(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.qk = torch.nn.Linear(784, 256)
        self.softmax = torch.nn.Softmax(-1)
 
    def forward(self, query, key, value):
        scale_factor  = 0.3  # Scaling factor
        inv_scale_factor  = scale_factor**-1
        vq  = self.qk(query).mul_(inv_scale_factor).transpose(-2, -1) 
        vqsoftmax  = self.softmax(vq) 
        dropout_vqsoftmax  = torch.nn.functional.dropout(vqsoftmax, p=0.5) 
        vs  = value
        vo  = dropout_vqsoftmax @ vs 
        return vo
```
