
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear = torch.nn.Linear(3, 8)
 
    def forward(self, x1, other=None):
        if other is None:
            # Use the input tensor as another argument to `forward`
            return self.linear(x1)
        else:
            v1 = self.linear(x1) + other
            v2 = torch.relu(v1)
            return v2
```
