
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear = torch.nn.Linear(256, 10)
 
    def forward(self, x):
        v1 = self.linear(x)
        v2 = (v1 - 1) * 3 / 2
        return v2
