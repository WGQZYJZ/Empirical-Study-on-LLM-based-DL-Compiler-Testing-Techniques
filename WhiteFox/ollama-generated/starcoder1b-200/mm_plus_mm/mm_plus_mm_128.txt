
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
 
    def forward(self, x1, x2, y3):
        v1 = torch.mm(x1, x2) # Matrix multiplication between the input tensors
        v2 = torch.mm(x3, y3) # Matrix multiplication between the input tensors and the output of a second model
        return v1 + v2
