# Model
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv = torch.nn.Conv2d(3, 8, kernel_size=1, stride=1, padding=1)
        self.linear = torch.nn.Linear(10, 40)

    def forward(self, x1):
        v1 = self.conv(x1)
        v2 = torch.nn.functional.relu(v1)
        v3 = self.linear(v2)
        return v3


# Initializing the model
m = Model()
