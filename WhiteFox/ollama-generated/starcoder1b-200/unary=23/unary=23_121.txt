
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv = torch.nn.Conv2d(8, 3, 1, stride=1, padding=1)
        self.transposed_conv = torch.nn.ConvTranspose2d(
            3, 8, kernel_size=(1, 3), stride=(1, 1), padding=(0, 0))
 
    def forward(self, x1):
        v1 = self.transposed_conv(self.conv(x1))
        v2 = torch.tanh(v1)
        return v2
```
