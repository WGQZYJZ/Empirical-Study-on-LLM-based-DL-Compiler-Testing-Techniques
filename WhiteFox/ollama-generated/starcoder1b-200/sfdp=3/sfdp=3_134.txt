
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv1 = torch.nn.Conv2d(3, 8, 1, stride=1, padding=1)
        self.conv2 = torch.nn.Conv2d(8, 16, 1, stride=1, padding=1)

    def forward(self, x1, x2):
        # Note that the input tensors need to be transposed because of how torch.matmul works, since
        #   qk has shape (B x D_k, K), while x is supposed to have shape (D_k, B). The transpose
        #   makes sure that the dimensions remain correct for a valid dot product.
        qk = torch.matmul(x1, x2.transpose(-2, -1))  # Compute the dot product of the input tensors
        scaled_qk = qk.mul(scale)  # Scale the dot product by a factor
        softmax_qk = scaled_qk.softmax(dim=-1)  # Apply softmax to the scaled dot product
        dropout_qk = torch.nn.functional.dropout(softmax_qk, p=dropout_p)  # Apply dropout to the softmax output
        output = dropout_qk.matmul(value)  # Compute the dot product of the dropout output and the value tensor
