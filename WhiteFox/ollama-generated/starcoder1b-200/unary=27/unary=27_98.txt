
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv = torch.nn.Conv2d(3, 8, 1, stride=1, padding=1)
 
    def forward(self, x1, min_value=0.0, max_value=None):
        v1 = self.conv(x1)
        v2 = v1 * 0.5
        if max_value is not None:
            v3 = torch.clamp_max(v2, max_value)
        else:
            v3 = torch.clamp_min(v2, min_value)
        return v3


# Initializing the model
m = Model()
```
The keyword arguments `min_value` and `max_value` are both required and default to zero if not specified.
