
class Model(torch.nn.Module):
    def __init__(self, min_value=-100.0, max_value=100.0):
        super().__init__()
        self.conv = torch.nn.Conv2d(3, 8, 1, stride=1, padding=1)

    def forward(self, x1):
        v1 = self.conv(x1)
        return v1 * torch.clamp_min(torch.clamp_max(v1, min_value), max_value)
