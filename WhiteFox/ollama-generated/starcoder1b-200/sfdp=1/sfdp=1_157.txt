
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear = torch.nn.Linear(d_model, d_inner)
 
    def forward(self, x1):
        qk = self.linear(x1).matmul(x2) # Compute the dot product of the query and key tensors
        v = qk.div(inv_scale_factor).softmax(dim=-1)  # Apply softmax to the scaled dot product
        o = torch.matmul(v, x2) # Compute the dot product of the dropout output and the value tensor
        return self.linear(o)


# Initializing the model
m = Model()
