
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.fc = torch.nn.Linear(10, 5)
 
    def forward(self, x1):
        qk = torch.matmul(x1, self.fc.weight.t())  # Compute the dot product of a query and key tensor
        inv_scale_factor = (
            (torch.pow(qk.exp(), -beta_2)).mul_(
                torch.pow((1 - beta_2), n_layer)
            ).mul_(kernel_size).sqrt_()
        )
        softmax_qk = qk / (inv_scale_factor.sqrt() * self.fc.weight.norm(dim=0))  # Apply softmax to the scaled dot product
        output = torch.nn.functional.dropout(softmax_qk, p=dropout_p)  # Apply dropout to the softmax output
        return output
