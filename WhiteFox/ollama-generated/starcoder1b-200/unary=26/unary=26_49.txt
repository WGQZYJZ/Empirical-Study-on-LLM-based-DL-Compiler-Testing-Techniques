
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv_transpose = torch.nn.ConvTranspose2d(8, 3, 1, stride=1, padding=1)
 
    def forward(self, x1):
        v1 = self.conv_transpose(x1) > 0
        v2 = v1 * -0.5
        v3 = v1 * -0.7071067811865476
        v4 = torch.where(v2, x1, v3)
        return v4
