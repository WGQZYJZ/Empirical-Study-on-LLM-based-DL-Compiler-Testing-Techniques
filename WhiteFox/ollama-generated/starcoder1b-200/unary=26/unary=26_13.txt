
class Model(nn.Module):

        self.conv = Conv2dTranspose(3, 8, 4, stride=2, padding=0)
        self.negative_slope = negative_slope
 
    def forward(self, x1):

        positive_part = torch.where(mask, x1, torch.zeros_like(x1))
        neg_mask = ~mask # Create a mask where each element is True if the corresponding element in t1 is greater than 0, False otherwise
        negative_part = negative_slope * self.conv(torch.ones_like(positive_part), mask)
