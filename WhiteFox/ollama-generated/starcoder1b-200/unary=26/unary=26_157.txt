
class Model(torch.nn.Module):
    def __init__(self, negative_slope=0.5):
        super().__init__()
        self.conv = torch.nn.ConvTranspose2d(8, 3, 1) # Transposed convolution followed by pointwise convolution with kernel size 1
        self.negative_slope = negative_slope
 
    def forward(self, x1):
        v1 = self.conv(x1)
        mask = (v1 > 0).type(torch.LongTensor) * self.negative_slope # Create a mask where each element is True if the corresponding element in v1 is greater than `0`, False otherwise
        v2 = v1  * mask.type(torch.DoubleTensor) # Multiply the output of the transposed convolution by the negative slope based on the mask
        return torch.where(mask, x1, v2).clamp_min_(0.)
```
