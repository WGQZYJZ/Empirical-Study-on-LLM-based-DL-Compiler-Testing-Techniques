
class Model(torch.nn.Module):
    def __init__(self, negative_slope=0.25):
        super().__init__()
        self.conv = torch.nn.ConvTranspose2d(3, 4, 2)
        self.negative_slope = negative_slope
 
    def forward(self, x1):
        v1 = self.conv(x1) > 0
        v2 = v1 * self.negative_slope
        return v2
