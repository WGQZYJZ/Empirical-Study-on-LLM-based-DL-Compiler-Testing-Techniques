
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv = torch.nn.ConvTranspose2d(8, 3, 1, stride=1, padding=1)
 
    def forward(self, x):
        v = self.conv(x)
        v -= 3
        v = torch.clamp_min(v, 0)
        v = torch.clamp_max(v, 6)
        return v / 6


# Initializing the model
m = Model()
