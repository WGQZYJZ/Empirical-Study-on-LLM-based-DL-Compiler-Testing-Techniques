
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear = torch.nn.Linear(10, 2)
 
    def forward(self, x1):
        v1 = self.linear(x1) + torch.randn(10, requires_grad=True).view(-1) * 0.5
        return v1
```
```
m = Model()

 # Inputs to the model
x1 = torch.randn(20, 10)
v1 = m(x1)
