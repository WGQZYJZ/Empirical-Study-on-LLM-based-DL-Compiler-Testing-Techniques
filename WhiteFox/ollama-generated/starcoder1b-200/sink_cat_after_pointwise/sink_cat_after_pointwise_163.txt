
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear = torch.nn.Linear(2, 2)

    def forward(self, x1):
        v1 = x1.permute(0, 2, 1)
        v2 = torch.cat([v1, v1], dim=0).view(-1) # Reorder the elements in 't2' according to their original positions (i.e., in 'v1')
        return torch.relu(self.linear(v2))
