
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear = torch.nn.Linear(2, 2)

    def forward(self, x1):
        v1 = x1.permute(0, 2, 1) # Erase the last two dimensions of x1 and permute it to (batch_size, input_channels, feature_dimension).
        v2 = self.linear(v1) # Invoke a linear transformation on v1 using self.linear.weight and self.linear.bias, respectively.

        return v2

# Initializing the model
m  = Model()
