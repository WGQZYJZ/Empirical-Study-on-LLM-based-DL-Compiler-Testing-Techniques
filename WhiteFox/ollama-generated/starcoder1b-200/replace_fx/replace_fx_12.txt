
class Model(torch.nn.Module):
    def __init__(self, random: bool = True):
        super().__init__()
        self.dropout  = torch.nn.functional.dropout

    def forward(self, x1, random=True):
        v1 = self.dropout(x1)
        v2 = x1 * random
        return v2
