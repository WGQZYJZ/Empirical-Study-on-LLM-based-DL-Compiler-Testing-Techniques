
class Model(torch.nn.Module):
    def __init__(self, dropout: bool = True):
        super().__init__()
        self.dropout = dropout

    def forward(self, x1):
        if self.dropout:
            v1 = torch.nn.functional.dropout(x1, p=0.25)
        else:
            v1 = x1

        v2 = torch.nn.functional.linear(v1, ...) # ... = `torch.nn.functional.dropout` or `torch.rand_like` function
        return v2

# Initializing the model
m = Model(False)  # `False` here indicates that dropout is disabled
```

The `m` object should be initialized with two arguments: `True` for `dropout=True`, and `False` otherwise.
