
class Model(torch.nn.Module):
    def __init__(self, config={}):
        super().__init__()
        self.linear = torch.nn.Linear(2, 2)

        # Fallbacks to the `lowmem_dropout` function.
        if config['fallback_random']:
            self.lowmem_dropout = LowMemDropout(config=config)

    def forward(self, x1):
        v1 = self.lowmem_dropout(x1)  # Erase original nodes invoking this function.
        v2 = torch.nn.functional.linear(v1, self.linear.weight, self.linear.bias)
        return v2


# Initializing the model
