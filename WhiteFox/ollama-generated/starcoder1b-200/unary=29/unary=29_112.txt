
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv = torch.nn.ConvTranspose2d(8, 3, 1, stride=1, padding=1)
 
    def forward(self, x1, min_value=0, max_value=1):
        v1 = self.conv(x1)
        v2 = v1  - torch.clamp(min_value + torch.clamp(max_value * v1, max_value, 0), min=0)
        return v2
