
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv = torch.nn.ConvTranspose2d(3, 8, 1, stride=1, padding=0)
 
    def forward(self, x1, min_value=-2.0, max_value=4.0):
        return self.conv(x1, min_value, max_value)


# Initializing the model
m = Model()
