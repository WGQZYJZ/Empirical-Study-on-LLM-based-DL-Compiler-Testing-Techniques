
class Model(torch.nn.Module):
    def __init__(self, min_value=1, max_value=100):
        super().__init__()
        self.conv = torch.nn.ConvTranspose2d(3, 8, 4, stride=2, padding=1)
 
    def forward(self, x1, min_value=1, max_value=100):
        v1 = self.conv(x1, min_value=min_value, max_value=max_value)
        return torch.clamp_min(v1, min_value), torch.clamp_max(v1, max_value)


# Initializing the model
m  = Model()


# Inputs to the model
x1  = torch.randn(1, 3, 64, 64)
v1, v2 = m(x1, min_value=0, max_value=50)
