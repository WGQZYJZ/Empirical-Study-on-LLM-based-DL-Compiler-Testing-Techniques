
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()

    def forward(self, x1, x2):  # x1 and x2 are both input tensors with more than two dimensions
        v1 = x1.permute(0, 2, 1)  # Permute the inputs so that t1 is at first dimension
        v2 = torch.bmm(v1, x2)  # or torch.matmul(v1, x2)
        return v2
```
