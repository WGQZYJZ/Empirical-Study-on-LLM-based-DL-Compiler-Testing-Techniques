
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear1 = torch.nn.Linear(2, 4)
        self.linear2 = torch.nn.Linear(4, 4)

    def forward(self, x1, x2):
        v1 = x1.permute(0, 2, 1)
        v2 = torch.bmm(v1, self.linear1.weight)  # or v2 = torch.matmul(v1, self.linear1.weight)
        return v2

# Initializing the model
m = Model()
