
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv = torch.nn.Conv2d(3, 8, 1, stride=1, padding=1)
 
    def forward(self, x1):
        v1 = self.conv(x1)
        split_sizes = [torch.tensor([i]) for i in range(v1.shape[1])]
        concatenated_tensor = torch.cat([torch.split(v1, sizes=split_sizes)[i] for i in range(len(split_sizes))], dim=0)
        return concatenated_tensor


# Initializing the model
m = Model()
