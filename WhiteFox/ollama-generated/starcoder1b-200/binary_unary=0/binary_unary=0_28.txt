
class Model(torch.nn.Module):
    def __init__(self, m: torch.nn.Module):
        super().__init__()
        self.conv = torch.nn.Conv2d(3, 8, 1, stride=1, padding=1)
        self.relu = torch.nn.ReLU()
        self.m  = m

    def forward(self, x1):
        v1 = self.conv(x1)
        v2 = self.relu(v1) + self.m
        return v2
