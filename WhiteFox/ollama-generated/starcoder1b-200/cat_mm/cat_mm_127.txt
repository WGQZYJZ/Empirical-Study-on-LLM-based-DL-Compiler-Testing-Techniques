
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv = torch.nn.Conv2d(3, 8, 1, stride=1, padding=1)
 
    def forward(self, x1, x2):
        v1 = self.conv(x1)
        v2 = torch.cat([v1, v1], dim=-1) # Concatenation of the result tensor along a dimension with value -1
        return v2


# Inputs to the model
input1  = torch.randn(3, 64, 64, requires_grad=True)
input2  = torch.randn(3, 64, 64, requires_grad=True)
__output__  = Model()(input1, input2)


# Input to the model and output of the forward function must be a tuple
m = Model()
x1, x2 = m.forward(torch.randn(1, 3, 64, 64), torch.ones_like(input1))
print(__output__) # __output__ is equal to input2
