
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.query = torch.nn.Linear(12, 64)
        self.key = torch.nn.Linear(12, 64)
        self.value = torch.nn.Linear(12, 64)
        self.attn_mask = None
 
    def forward(self, x):
        # Compute dot product of x with query and key (and then scale by sqrt(size(-1)))
        qk = x @ self.query  # [B, Q] * [C, K] = [B, C]
        qk  = torch.softmax(qk, dim=-1)  # [B, Q, K] -> [B, Q, K] / sqrt([B, C])
        qk  = torch.dropout(qk, dropout_p, True)  # Apply dropout on [B, Q, K]

        value  = x @ self.value  # [B, V] * [C, K] = [B, C]
        attn_mask  = torch.eye(x.size(0)).unsqueeze(-1)  # [B, 1, 1, H, W]
        attn_mask  = attn_mask.repeat((1, x.size(1), 1, 1, x.size(2)))  # [B, N, H, W, C] = [B, H*W, C]
        attn_weight = qk @ value  # [B, N, H, W, C] * [B, K, V] = [B, K, V]

        return attn_weight @ self.value
```
This pattern characterizes scenarios where the query, key, and value are all identical to each other (and then their dot products are computed separately), followed by a dropout operation on the attention weights.
