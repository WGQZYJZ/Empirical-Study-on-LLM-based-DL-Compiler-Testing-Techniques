
class Model(torch.nn.Module):
    def __init__(self, negative_slope: float = 0.01):
        super().__init__()
        self.conv = torch.nn.Conv2d(3, 8, 1, stride=1, padding=1)
 
    def forward(self, x1):
        v1 = self.conv(x1)
        v2 = torch.where(v1 > 0, v1 * negative_slope, x1)  # The output of the convolution is multiplied by a constant based on t2
        return v2


# Initializing the model
m = Model()

x1 = torch.randn(1, 3, 64, 64)
