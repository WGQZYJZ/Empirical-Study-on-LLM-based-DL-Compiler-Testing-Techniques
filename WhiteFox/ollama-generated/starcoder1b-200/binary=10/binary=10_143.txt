
class Model(torch.nn.Module):
    def __init__(self, other_tensor):
        super().__init__()
        self.linear = torch.nn.Linear(3, 8)
        self.other = other_tensor
 
    def forward(self, x1):
        v1 = self.linear(x1)
        return v1 + self.other
```
# Initializing the model
m = Model(torch.randn(1))
```
