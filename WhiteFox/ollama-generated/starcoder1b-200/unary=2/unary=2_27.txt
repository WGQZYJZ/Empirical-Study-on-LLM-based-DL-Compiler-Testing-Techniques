
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv = torch.nn.ConvTranspose2d(3, 8, 1, stride=1, padding=1)
 
    def forward(self, x1):
        v1 = self.conv(x1) * 0.5
        v2 = v1 * v1 * v1
        v3 = v2 * 0.044715
        v4 = v1 + v3
        v5 = v4 * 0.7978845608028654
        v6 = torch.tanh(v5) + 1
        return v6


# Initializing the model
m = Model()
