
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv  = torch.nn.Conv2d(3, 8, 1)
        self.transpose = torch.nn.ConvTranspose2d(8, 3, 1)
 
    def forward(self, x1):
        v1 = self.conv(x1) * 0.5
        v2 = v1  + 1 # Add 1 to the output of the convolution

        v3 = v2  * v2  * v2  * v2  * v2  * v2  * v2  * v2  # Multiply the cubed output by 0.044715
        v4 = torch.tanh(v3)

        v5 = v1 + v4
        v6 = v5  * v5  # Cube the output of the addition
        v7 = self.transpose(v6)
        return v7


# Initializing the model
m = Model()
