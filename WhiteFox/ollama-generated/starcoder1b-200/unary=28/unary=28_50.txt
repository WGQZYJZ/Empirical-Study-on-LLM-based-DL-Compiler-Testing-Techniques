
class Model(torch.nn.Module):
    def __init__(self, min_value=0., max_value=1.,):
        super().__init__()
        self.linear = torch.nn.Linear(in_features=64*3*3*8, out_features=32)
 
    def forward(self, x):
        v  = self.linear(x)
        v = torch.clamp(v, min_value=min_value, max_value=max_value)
        return v


# Initializing the model
m = Model(min_value=-1., max_value=1.)
