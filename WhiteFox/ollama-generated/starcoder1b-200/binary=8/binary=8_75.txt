
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv = torch.nn.Conv2d(3, 8, 1, stride=1, padding=1)

    def forward(self, x1: torch.Tensor, other: torch.Tensor) -> torch.Tensor:
        v1 = self.conv(x1)
        v2 = v1 + other
        return v2
