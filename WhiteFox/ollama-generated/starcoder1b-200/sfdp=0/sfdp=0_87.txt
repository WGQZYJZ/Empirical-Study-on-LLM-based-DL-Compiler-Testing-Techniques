
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv  = torch.nn.Conv2d(3, 8, 1)
        self.query_layer = torch.nn.Linear(8, 16)
        self.key_layer = torch.nn.Linear(8, 16)
 
    def forward(self, x):
        query = self.query_layer(x)
        key = self.key_layer(x)
        scaled_dot_product = torch.matmul(query, key.transpose(-2, -1)) / math.sqrt(8 * 8)
        attention_weights = scaled_dot_product.softmax(dim=-1)
        output = attention_weights.matmul(x)
        return output
