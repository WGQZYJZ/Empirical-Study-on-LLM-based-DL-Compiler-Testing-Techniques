
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.fc = torch.nn.Linear(1024, 10)
 
    def forward(self, x1):
        x1 = x1.view(-1, 1024) # Reshape from [batch_size * seq_len, input_dim] to [batch_size * seq_len, -1]
        v1 = self.fc(x1)
        attention_weights = torch.softmax(v1, dim=-1) # Apply the softmax function on the last dimension to compute attention weights
        output = attention_weights.matmul(v1)
        return output


# Initializing the model
m  = Model()
