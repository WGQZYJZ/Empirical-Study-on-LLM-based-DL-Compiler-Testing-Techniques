
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv = torch.nn.Conv2d(3, 8, 1, stride=1, padding=1)

    def forward(self, x1, x2):
        v1 = self.conv(x1)
        v2 = self.conv(x2)
        scale = torch.sqrt((v1 ** 2 + v2 ** 2).view(-1))
        scaled_dot_product = torch.matmul(v1.view(-1), v2.view(-1)).div(scale)
        attention_weights = scaled_dot_product.softmax(dim=-1)
        output = attention_weights.matmul(x2.view(-1).reshape((-1, 1)))

        return output
