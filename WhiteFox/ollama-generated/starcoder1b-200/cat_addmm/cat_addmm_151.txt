
class Model(torch.nn.Module):
    def __init__(self, hidden_units=16):
        super().__init__()
        self.fc = torch.nn.Linear(28 * 28, hidden_units)
 
    def forward(self, x1):
        v1 = self.fc(x1)
        return v1


# Initializing the model
m = Model()


# Inputs to the model
x1  = torch.randn(100, 28 * 28)
__output__  = m(x1)
