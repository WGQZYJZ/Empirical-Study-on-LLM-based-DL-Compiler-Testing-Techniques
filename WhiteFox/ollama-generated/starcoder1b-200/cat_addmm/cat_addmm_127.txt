
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv = torch.nn.Conv2d(3, 8, 1, stride=1, padding=1)
 
    def forward(self, x1, x2):
        mat1 = torch.randn((1, 3, 4, 4))
        mat2 = torch.randn((1, 8, 3, 3))
        v1 = self.conv(x1)
        v2 = torch.addmm(v1, mat1, mat2)
        v3 = torch.cat([v2], dim=1)
        return v3
