
class Attention(torch.nn.Module):
    def __init__(self, in_dim, out_dim=None, dropout_rate=0.2):
        super().__init__()
        self.dropout = torch.nn.Dropout(p=dropout_rate)
        self.in_proj = torch.nn.Linear(in_dim, out_dim or in_dim)
        self.out_proj = torch.nn.Linear(out_dim or in_dim, in_dim)
 
    def forward(self, x):
        h  = self.dropout(x)
        h  = F.leaky_relu_(self.in_proj(h), negative_slope=0.1)
        att  = F.softmax(self.out_proj(h).squeeze(-1), dim=-1) * h
        output  = (att @ self.dropout(x)).transpose(-2, -1) + x
        return output


class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.layer_norm_before = torch.nn.LayerNorm(normalized_shape=512, elementwise_affine=True)
        self.layer_norm_after  = torch.nn.LayerNorm(normalized_shape=512, elementwise_affine=True)
        self.attention0 = Attention(out_dim=1024)
 
    def forward(self, x):
        layer_input  = self.layer_norm_before(x)
        attention0  = self.attention0(layer_input)
        attention0  = self.dropout(attention0)
        output  = (layer_input + attention0).transpose(-2, -1)
        output  = self.layer_norm_after(output)
        return output
