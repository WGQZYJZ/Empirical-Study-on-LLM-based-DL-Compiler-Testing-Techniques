
class ScaledDotProductAttention(torch.nn.Module):
    def __init__(self, dim_key: int, dim_value: int, scale=None):
        super().__init__()
        self.dim_key = dim_key
        self.dim_value = dim_value
 
        if scale is None:
            inv_sqrt_dim_key = 1.0 / math.sqrt(self.dim_key)
        else:
            assert type(scale) == float
            inv_sqrt_dim_key = 1.0 / (math.sqrt(self.dim_key) * scale)
 
        self.scale_factor = torch.tensor([inv_sqrt_dim_key])
 
    def forward(self, query, key, value):
        dim_key = query.size(-2)
        scaled_dot_product = torch.matmul(query, key.transpose(-2, -1)) * self.scale_factor[None]
        attention_weights = scaled_dot_product.softmax(dim=-1)
 
        return attention_weights.matmul(value), attention_weights
