
class TransformerEncoder(torch.nn.Module):
    def __init__(self, embedding_dim=64, num_heads=8, dim_feedforward=2048):
        super().__init__()
 
        self.layernorm1 = torch.nn.LayerNorm(embedding_dim)
        self.self_attn1 = MultiheadAttention(
            embedding_dim, num_heads, dropout=0.0,
            attention_dropout=0.0
        )
 
        self.layernorm2 = torch.nn.LayerNorm(embedding_dim)
        self.linear1 = torch.nn.Linear(embedding_dim, dim_feedforward)
 
        self.activation = torch.nn.ReLU()
        self.linear2 = torch.nn.Linear(dim_feedforward, embedding_dim)
 
    def forward(self, x):
        attn_output = self.self_attn1(x)
        attn_output = self.layernorm1(x + attn_output)
 
        out = self.linear2(self.activation(self.linear1(attn_output)))
        out = self.layernorm2(attn_output + out)
        return out
