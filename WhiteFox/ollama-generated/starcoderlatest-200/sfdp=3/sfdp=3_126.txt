
class Attention(torch.nn.Module):

    def forward(self, query, key, value, scale_factor=None):
        # Scale the dot product by a factor and apply softmax to it 
        scaled_qk = torch.matmul(query, key.transpose(-2, -1)) * (scale_factor or 1)
        softmax_qk = scaled_qk.softmax(dim=-1)
        dropout_qk = F.dropout(softmax_qk, p=0.5)
 
        # Compute the dot product of the dropout output and the value tensor 
        return torch.matmul(dropout_qk, value).transpose(-2, -1)
