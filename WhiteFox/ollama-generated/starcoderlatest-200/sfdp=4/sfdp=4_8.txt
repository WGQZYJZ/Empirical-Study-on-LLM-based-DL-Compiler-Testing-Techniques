
class Attention(torch.nn.Module):
    def __init__(self, dim, num_heads = 1, attn_mask=None):
        super().__init__()
        self.num_heads = num_heads
        assert (dim % self.num_heads == 0)
        self.attn_mask = attn_mask
 
        # Create query and key layers
        self.query_layer = torch.nn.Linear(dim, dim * self.num_heads)
        self.key_layer   = torch.nn.Linear(dim, dim * self.num_heads)
 
        # Create value layer
        self.value_layer  = torch.nn.Linear(dim, dim)
 
    def forward(self, x):
        batch_size = x.shape[0]
        d = x.shape[-1]
 
        # Query vector: (batch_size * num_heads, q_length, dim_per_head)
        query = self.query_layer(x).view(batch_size, -1, self.num_heads, d // self.num_heads)
        key   = self.key_layer(x).view(batch_size, -1, self.num_heads, d // self.num_heads)
 
        # Attention weights: (batch_size * num_heads, q_length, v_length)
        attn_weights  = torch.matmul(query, key.transpose(-2, -1)) / math.sqrt(dim)
        if attn_mask is not None:
            attn_weights += attn_mask
 
        # Attention output: (batch_size * num_heads, q_length, d)
        v = torch.matmul(attn_weights, self.value_layer(x).view(batch_size, -1, dim))
        out = v.transpose(-2, -1).contiguous().view(batch_size, -1, d)
 
        return out
