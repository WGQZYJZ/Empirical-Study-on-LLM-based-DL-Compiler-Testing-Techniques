
class QueryKeyAttentionModel(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.attn = torch.nn.MultiheadAttention()
 
    def forward(self, query, key, value, attn_mask):
        v1, attn_weight = self.attn(query, key, value, attn_mask=attn_mask)  # Apply multi-head attention to the query and key, and store the output in v1, and also store the attention weights in attn_weights
        return v1
