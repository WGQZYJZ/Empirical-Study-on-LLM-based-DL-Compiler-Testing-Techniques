
class Model(torch.nn.Module):

    def forward(self, x1):
        v1 = self.conv(x1)
        v2 = v1 * 0.5
        v3 = v1 ** 2 * v1
        v4 = v3 * 0.00928947442826733
        v5 = v1 + v4
        v6 = v5 * 0.7955591490471178
        v7 = torch.tanh(v6)
