
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear1 = torch.nn.Linear(2, 2)

    def forward(self, x1):
        v1 = x1.permute(0, 2, 3) # Swap the last two dimensions of input tensor
        t1 = self.linear1(v1)
        return t1
```
