
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear = torch.nn.Linear(2, 1)

    def forward(self, x1):
        v1 = torch.cat([x1, x1], dim=1).permute(0, 3, 1, 2) # concatenate the last two dimensions of input tensor with itself and permute to make them compatible for the linear transformation
        v2 = torch.nn.functional.linear(v1, self.linear.weight, self.linear.bias)
        return v2
