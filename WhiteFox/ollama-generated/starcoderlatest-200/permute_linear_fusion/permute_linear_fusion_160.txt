
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear = torch.nn.Linear(2, 3)

    def forward(self, x1):
        v1 = torch.nn.functional.avg_pool2d(x1, (4, 4))
        # Add random noise to the input tensor
        v2 = v1 + torch.randn_like(v1)
        t2 = self.linear(v2)
        return t2
```


# Initializing the model
m = Model()
