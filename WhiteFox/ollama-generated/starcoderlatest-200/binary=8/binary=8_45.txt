
class Model(torch.nn.Module):
    def __init__(self, other=None):
        super().__init__()
        if other == None:
            self.conv = torch.nn.Conv2d(3, 8, 1, stride=1, padding=1)
        else: 
            self.conv = torch.nn.Conv2d(3, 8, 1, stride=1, padding=1)
            self.other_tensor = other
    def forward(self, x1):
        v1 = self.conv(x1)
        if 'other_tensor' in dir(self):
            return v1 + self.other_tensor
        else: 
            return v1
 
 # Initializing the model
m = Model()

 # Inputs to the model
x1 = torch.randn(1, 3, 64, 64)
v2 = m(x1) ## Input the tensor you want to be added as a keyword argument: v2 = m(x1, other=other_tensor)
