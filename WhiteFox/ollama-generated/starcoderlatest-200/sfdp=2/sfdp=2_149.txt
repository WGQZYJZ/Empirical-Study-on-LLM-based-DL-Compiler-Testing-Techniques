
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
 
        self.attn_q = torch.nn.Linear(embed_dim, hidden_size)  # Linear layer that computes query for multi-head attention
        
        self.attn_k = torch.nn.Linear(embed_dim, hidden_size)  # Linear layer that computes key for multi-head attention
        self.attn_v = torch.nn.Linear(embed_dim, hidden_size)  # Linear layer that computes value for multi-head attention
        
        self.fc_q = torch.nn.Linear(hidden_size * num_heads, hidden_size)  # Linear layer after multi-head attention
        self.fc_v = torch.nn.Linear(hidden_size * num_heads, hidden_size)
 
        self.dropout_q = torch.nn.Dropout(p=self.dropout_p)  # Apply dropout to the query before multi-head attention
        self.dropout_k = torch.nn.Dropout(p=self.dropout_p)  # Apply dropout to the key before multi-head attention
        self.dropout_v = torch.nn.Dropout(p=self.dropout_p)  # Apply dropout to the value before multi-head attention
        
    def forward(self, q, k, v):
        batch_size = q.shape[0]
 
        qk = torch.cat([q.unsqueeze(1).expand(-1, num_heads, -1, -1), 
                        k.unsqueeze(2).expand(-1, num_heads, -1, -1)], dim=3)  # Concatenate the query with all keys
        v = self.attn_v(v).permute(0, 2, 1, 3).contiguous().view(batch_size * num_heads, embed_dim, -1)  # Extract and reshape the value for each head from the feature map
        
        qk = torch.matmul(qk, self.attn_q(qk)).permute(0, 2, 3, 1).contiguous().view(batch_size * num_heads, -1, embed_dim)  # Compute the dot product of the query and the key for each head
        scaled_qk = qk.div(self.scale_factor)  # Scale the dot product by the scale factor
        
        softmax_qk = scaled_qk.softmax(dim=-1)  # Apply softmax to the scaled dot product
        dropout_qk = self.dropout_q(softmax_qk)  # Apply dropout to the softmax output
 
        out_v = torch.matmul(dropout_qk, v)  # Compute the attention weights on all features in the value tensor for each head and concatenate them
        out_v = out_v.permute(0, 2, 1, 3).contiguous().view(batch_size, num_heads, -1, embed_dim)
        
        v_proj = self.fc_v(out_v).permute(0, 2, 1, 3).contiguous().view(batch_size * num_heads, -1, hidden_size)
        
        v_q_concat = torch.cat([v_proj, q.unsqueeze(1)], dim=1)
        out_q = self.fc_q(v_q_concat).permute(0, 2, 1, 3).contiguous().view(batch_size * num_heads, -1, hidden_size)
 
        return out_q
 
# Initializing the model and printing its parameters' shape
m = Model()
print("Parameter Shapes:")

    print("%s : %s" %(name, str(param.shape)))

value = torch.randn(batch_size, num_heads * head_dim, embed_dim)  # Input data for the value tensor of shape (batch_size, num_heads * head_dim, embed_dim)


# Forward pass: compute predicted outputs by passing inputs to the model
out = m(query, key, value)
