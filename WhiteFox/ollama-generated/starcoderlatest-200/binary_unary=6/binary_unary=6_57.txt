
class Model2(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv = torch.nn.Conv2d(3, 8, 1, stride=1, padding=1)
 
    def forward(self, x1):
        v1 = self.conv(x1)
        v2 = v1 - 0.1 # Subtract 0.1 from the output of the linear transformation
        v3 = torch.nn.ReLU()(v2)
        return v3
