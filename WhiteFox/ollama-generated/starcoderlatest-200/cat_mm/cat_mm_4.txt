
class Model(torch.nn.Module):
    def __init__(self, n):
        super().__init__()
        self.conv = torch.nn.Conv2d(3 * n, 8, 1)
 
    def forward(self, x1):
        v1 = self.conv(x1.view(-1, 3 * n))
        v2 = v1 * 0.5
        v2 = v2 * 0.7071067811865476
        v2 = torch.erf(v2)
        v2 = v2 + 1
        v2 = v2 * 0.5
        return v2
