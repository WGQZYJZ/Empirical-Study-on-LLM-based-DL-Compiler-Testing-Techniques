
class Model(torch.nn.Module):
    def __init__(self, min_value, max_value=100):
        super().__init__()
        self.linear = torch.nn.Linear(64 * 64, 8)
 
    def forward(self, x1):
        v1 = self.linear(x1.reshape(-1))
        v2 = torch.clamp_min(v1, min=min_value)
        v3 = torch.clamp_max(v2, max=max_value)
        return v3


# Initializing the model and providing keyword arguments
m = Model(min_value=-50)
m2 = Model(-50)
__output1__ = m(x1)  # -42.76241180419922
__output2__ = m2(x1)  # -50.27392578125
