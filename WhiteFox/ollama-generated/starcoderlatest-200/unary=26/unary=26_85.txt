
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv_t = torch.nn.ConvTranspose2d(3, 8, 1, stride=1, padding=1)
 
    def forward(self, x1):
        v1 = self.conv_t(x1)
        negative_slope = 0.01
        mask = (v1 > 0).float()
        v2 = v1 * negative_slope
        v3 = torch.where(mask, v1, v2)
        return v3
```
