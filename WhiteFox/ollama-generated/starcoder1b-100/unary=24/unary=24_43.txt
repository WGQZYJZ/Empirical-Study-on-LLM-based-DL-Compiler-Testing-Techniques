
class Model(nn.Module):
    def __init__(self):
        super().__init__()
        self.conv = nn.Conv2d(3, 8, 1, stride=1, padding=1)
 
    def forward(self, x1):
        v1 = self.conv(x1).abs()
        v2 = torch.where(v1 > 0, v1, torch.zeros_like(v1)) * -0.5
        v3 = torch.mul(v1, 0.7071067811865476)
        v4 = torch.erf(v3).abs() + 1
        v5 = torch.mul(v2, v4) * -0.5
        v6 = v5 * v5
        return v6

# Initializing the model
m = Model()
