
class Model(torch.nn.Module):
    def __init__(self, min_value=0.9985615247608369, max_value=0.9986358093230887):
        super().__init__()
        self.conv = torch.nn.Conv2d(3, 8, 1, stride=1, padding=1)
 
    def forward(self, x1):
        v1 = self.conv(x1)
        v2 = v1.clamp_min(min_value).clamp_max(max_value)
        return v2
