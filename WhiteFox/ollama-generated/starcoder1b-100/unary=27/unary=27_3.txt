
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv = torch.nn.Conv2d(3, 8, 1, stride=1, padding=1)

    def forward(self, x1, min_value, max_value):
        v1 = self.conv(x1)
        v2 = v1 * (min_value - 0.5) + 0.5  # Subtract a constant from the output of the previous convolution
        v3 = v1 * (max_value - 0.5) + 0.5
        return v3
