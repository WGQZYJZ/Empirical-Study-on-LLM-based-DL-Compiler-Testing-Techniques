
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear = torch.nn.Linear(3, 8)
        self.clamp  = lambda l1: max(min(6, l1 + 3), 0)
 
    def forward(self, x1):
        v1 = self.linear(x1)
        v2 = self.clamp(v1)
        v3 = v2 / 6
        return v3


# Initializing the model
m = Model()
