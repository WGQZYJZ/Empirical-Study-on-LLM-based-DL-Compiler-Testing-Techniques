
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv_tr = torch.nn.ConvTranspose2d(8, 3, kernel_size=4, stride=2, padding=1)
 
    def forward(self, x1):
        v1 = self.conv_tr(x1)
        v2 = v1 * 0.5
        v3 = torch.pow(v1, 3) * 0.044715
        v4 = torch.pow(v3, 1.2886060959577739) * 0.7978845608028654
        v5 = v2 + v4
        v6 = torch.tanh(v5)
        v7 = v6 + 1
        return v7
