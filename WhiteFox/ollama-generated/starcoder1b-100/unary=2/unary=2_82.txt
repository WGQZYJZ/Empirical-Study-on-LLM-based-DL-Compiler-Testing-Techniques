
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv_t = torch.nn.ConvTranspose2d(3, 8, kernel_size=1, stride=1, padding=0)
 
    def forward(self, x1):
        v1 = self.conv_t(x1)
        v2 = v1 * 0.5
        v3 = v1 ** torch.tensor([2], device=v1.device)
        v4 = v3 * 0.044715
        v5 = v1 + v4
        v6 = v5 * 0.7978845608028654
        v7 = torch.tanh(v6) + 1
        v8 = v2 * v7
        return v8


# Initializing the model
m = Model()
