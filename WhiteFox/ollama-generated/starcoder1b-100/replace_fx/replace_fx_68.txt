
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()

    def forward(self, x1, t2=None):
        x  = torch.nn.functional.dropout(x1, p=0.8)
        t3 = torch.rand_like(x, dtype=torch.float64) if t2 is None else t2

        return x + t3
```
