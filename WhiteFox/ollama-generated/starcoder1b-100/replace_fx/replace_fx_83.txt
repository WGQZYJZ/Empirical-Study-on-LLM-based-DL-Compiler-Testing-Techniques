
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear  = torch.nn.Linear(2, 2)

    def forward(self, x1):
        # Dropout is always replaced by its corresponding replacement function (lowmem_dropout and rand_like respectively)
        v1 = torch.nn.functional.dropout(x1, p=0.5)  # The value of `p` parameter determines the probability of dropping some of the input tensors

        v2 = self.linear(v1)
        return v2


# Initializing the model
m = Model()
