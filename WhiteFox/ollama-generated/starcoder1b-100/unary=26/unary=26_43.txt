
class Model(nn.Module):
    def __init__(self, negative_slope=0.2):
        super().__init__()
        self.conv = nn.ConvTranspose2d(3, 8, kernel_size=3, stride=1, padding=1)
        self.negative_slope = negative_slope
 
    def forward(self, x1):
        mask = torch.gt(x1, 0).view(-1, 1)  # Create a mask where each element is True if the corresponding element in t1 is greater than 0, False otherwise
        mask = self.negative_slope * (mask - mask.float().sum(-1, keepdim=True))  # Multiply the output of the transposed convolution by the negative slope
        mask = torch.where(mask, x1, -x1)  # Apply the where function to select elements from t1 or t3 based on the mask t2
        return self.conv(mask)
