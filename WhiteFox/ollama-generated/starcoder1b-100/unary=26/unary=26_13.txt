
class Model(torch.nn.Module):
    def __init__(self, negative_slope=0.2):
        super().__init__()
        self.conv  = torch.nn.ConvTranspose2d(3, 8, 4, stride=1, padding=1)

    def forward(self, x1):
        v1 = self.conv(x1) > 0
        v2 = v1 * negative_slope
        v3 = torch.where(v1, v1, v2)
        return v3
```
