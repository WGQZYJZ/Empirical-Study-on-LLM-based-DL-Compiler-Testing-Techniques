
class Model(torch.nn.Module):
    def __init__(self, d_model):
        super().__init__()
        self.d_model = d_model
        self.w1  = torch.nn.Parameter(torch.FloatTensor(d_model, d_model))
        self.w2  = torch.nn.Parameter(torch.FloatTensor(d_model, d_model))
        self.b1  = torch.nn.Parameter(torch.FloatTensor(1))
        self.b2  = torch.nn.Parameter(torch.FloatTensor(1))
 
    def forward(self, x):
        v1  = self.w1 @ x
        v2  = self.w2 @ v1
        output  = v2 + self.b1
        return output


# Initializing the model
m  = Model(d_model=64)


# Inputs to the model
x1 = torch.randn(1, 3, 64, 64)
__output__  = m(x1)
