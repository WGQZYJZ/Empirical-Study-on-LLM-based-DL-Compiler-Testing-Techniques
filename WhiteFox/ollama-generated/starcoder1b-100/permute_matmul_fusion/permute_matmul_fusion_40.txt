
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear = torch.nn.Linear(2, 2)

    def forward(self, x1):
        v1  = x1.permute(0, 2, 1)
        v2  = torch.bmm(v1, self.linear.weight)
        return v2


# Initializing the model
m = Model()


# Inputs to the model
x1  = torch.randn(2, 3, 4, 5)
y1 = torch.randn(1, 4, 3, 7)
z1 = m(x1, y1)
