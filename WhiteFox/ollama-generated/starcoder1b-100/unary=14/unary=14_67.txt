
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv = torch.nn.ConvTranspose2d(3, 8, kernel_size=4, stride=1, padding=0)
        self.fc  = torch.nn.Linear(3, 1)
 
    def forward(self, x1):
        v1 = self.conv(x1)
        v2 = torch.sigmoid(v1)
        v3 = x1 * v2
        return v3


# Initializing the model
m = Model()
