
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv = torch.nn.Conv2d(3, 8, 1, stride=1, padding=1)
        self.attn = TransformerSelfAttention(self.vocab_size, 10)
 
    def forward(self, x1):
        v1 = self.conv(x1)
        qk = self.attn((v1, x1))
        attn_weight = torch.softmax(qk, dim=-1)
        value = self.attn.value(x1) * attn_weight
        return value


# Initializing the model
m  = Model()
