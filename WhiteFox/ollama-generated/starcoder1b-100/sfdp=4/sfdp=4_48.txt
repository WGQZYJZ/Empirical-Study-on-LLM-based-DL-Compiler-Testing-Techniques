
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv = torch.nn.Conv2d(3, 8, 1, stride=1, padding=1)
        self.layer_norm = torch.nn.LayerNorm([8])

    def forward(self, x1, x2):
        x = self.conv(x1)
        return (self.layer_norm(x) * x2).sum(dim=-1)
