
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear  = torch.nn.Linear(2, 2)

    def forward(self, x1):
        v1  = x1.permute(0, 2, 1)
        v2  = torch.nn.functional.convXd(v1, self.linear.weight, self.linear.bias, self.linear.stride,
                                          padding=self.linear.padding, dilation=self.linear.dilation, groups=self.linear.groups, ceil_mode=self.linear.ceil_mode)
        return v2
