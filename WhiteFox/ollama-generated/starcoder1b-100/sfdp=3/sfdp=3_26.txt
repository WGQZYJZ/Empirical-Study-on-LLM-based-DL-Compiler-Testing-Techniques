    def __init__(self):
        super().__init__()
        self.query = torch.nn.Linear(3, 4)
        self.key = torch.nn.Linear(3, 5)
        self.value = torch.nn.Parameter(torch.zeros((5, 6)))
 
    def forward(self, x1):
        # The input tensor to the model contains a batch size and sequence length dimension
        query = self.query(x1).view(-1, 3, 4)
        key = self.key(x1).view(-1, 3, 5)
        value = self.value.expand(-1, 5) * 0.7071067811865476
 
        # The dot product of the query and key tensors is scaled by a factor
        # so that the result of this operation is an unnormalized softmax over the same scale as the query and key tensors
        scaled_qk = torch.matmul(query, key.transpose(-2, -1))
        softmax_qk = scaled_qk.softmax(dim=-1)
 
        # The output of dropout layer of softmax is multiplied by a constant 0.7071067811865476
        # so that the result of this operation is an unnormalized pointwise multiplication of the query and key tensors
        scaled_value = value * 0.7071067811865476
        dropout_qk = torch.nn.functional.dropout(softmax_qk, p=dropout_p)
 
        # The output of pointwise operation (multiplication) is multiplied by a value tensor
        output = dropout_qk.matmul(scaled_value)
