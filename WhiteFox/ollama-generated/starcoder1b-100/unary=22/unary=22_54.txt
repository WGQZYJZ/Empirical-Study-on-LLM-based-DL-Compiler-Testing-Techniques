
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear = torch.nn.Linear(4, 2)
 
    def forward(self, x1):
        v1 = self.linear(x1)
        return torch.tanh(v1)


# Initializing the model
m = Model()

# Inputs to the model
x1 = torch.randn(2, 4, requires_grad=True)
y1 = m(x1)
y1.backward()
