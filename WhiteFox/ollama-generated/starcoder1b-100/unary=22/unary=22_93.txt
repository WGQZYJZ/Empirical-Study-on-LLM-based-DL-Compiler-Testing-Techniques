
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear = torch.nn.Linear(10, 1)
 
    def forward(self, x):
        v = self.linear(x)
        return v  # We don't care about the actual values of these tensors because they are generated by nn.Linear


# Initializing the model
m = Model()
