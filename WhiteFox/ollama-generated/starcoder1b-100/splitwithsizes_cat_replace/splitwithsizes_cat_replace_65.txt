
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv = torch.nn.Conv2d(3, 8, 1, stride=1, padding=1)
 
    def forward(self, x1):
        split_sizes = [1, 5] # Set sizes of two splits in the concatenation operation
        concatenated_tensor = torch.cat([torch.split(x1, split_sizes[i], dim=1)[i] for i in range(len(split_sizes))], dim=2) # Concatenate the split tensors along the same dimension
        return torch.is_finite(concatenated_tensor).type(torch.FloatTensor).unsqueeze(-1).squeeze(-1)  # Apply the condition to get output as a BooleanTensor


# Testing model against requirements
m = Model()

with torch.no_grad():
    for x1, expected in zip([x1, None], [True, False]):
        actual = m(x1)
        assert expected == actual
