
class Model(torch.nn.Module):

    def forward(self, x1, x2):
        # [1 x 8] x [3 x 64 x 64] -> [2 x 64 x 64]
        v1 = torch.cat([torch.split(x1, 2, dim=1)[0],
                       torch.split(x1, 2, dim=1)[1]], dim=1)
        # [1 x 8] x [3 x 64 x 64] -> [2 x 64 x 64]

        # [3 x 8] -> [1 x 8]
        v4 = torch.erf(v3)
        # [1 x 8] -> [2 x 8]

        # [2 x 8] -> [3 x 8]
        v6 = v2 * v5
        # [2 x 8] -> [1 x 8]
