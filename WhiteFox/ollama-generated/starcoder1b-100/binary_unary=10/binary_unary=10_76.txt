# Model
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv1 = torch.nn.Conv2d(3, 8, 1, stride=1, padding=1)
        self.conv2 = torch.nn.Conv2d(8, 16, 1, stride=1, padding=1)
        self.linear = torch.nn.Linear(8 * 4, 32)
 
    def forward(self, x1):
        v1 = self.conv1(x1)
        v2 = relu(v1)
        v3 = self.conv2(v2)
        return relu(self.linear(v3))
```
