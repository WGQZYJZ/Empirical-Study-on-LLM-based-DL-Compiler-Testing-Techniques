
class Model(torch.nn.Module):
    def __init__(self, min_value=0.5, max_value=10):
        super().__init__()
        self.linear  = torch.nn.Linear(3, 8)
        self.clamp    = lambda x, min_value, max_value: torch.max((x - min_value) / (max_value - min_value), 0)
 
    def forward(self, x1):
        v1  = self.linear(x1)
        v2 = self.clamp(v1, self.min_value, self.max_value)
        return v2


# Initializing the model
m = Model(50, 50)
