
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear = torch.nn.Linear(3, 4)
 
    def forward(self, x1):
        v1 = self.linear(x1)
        v2 = v1 + torch.ones((1, 4), dtype=torch.float32) # Add a dummy tensor to the output of the linear transformation
        return v2


# Initializing the model
m = Model()
