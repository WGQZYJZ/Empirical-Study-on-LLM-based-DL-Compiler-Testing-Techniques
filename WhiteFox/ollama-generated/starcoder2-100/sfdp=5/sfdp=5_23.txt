class Transformer(torch.nn.Module):
    def __init__(self, embedding_dim=512):
        super().__init__()

        self.position = torch.nn.Parameter(torch.rand((embedding_dim), 2))
        self.query = torch.nn.Linear(4, embedding_dim)
        self.key = torch.nn.Linear(640, embedding_dim)
        self.value = torch.nn.Linear(512, embedding_dim)
 
    def forward(self, x):

        pos_emb  = torch.sin(self.position[None, None, ...].mul(torch.arange(0., float(x.size(-3)) * float(embedding_dim), 2 * math.pi / float(embedding_dim))))
        query_out  = self.query(pos_emb)
        
        x  = x @ query_out  # Compute the dot product of the positional embedding and the query
        key, value = torch.chunk(x, chunks=3, dim=-1)

        attn_mask  = torch.zeros((key.size(-2), key.size(-1))).float().masked_fill_(torch.eye(key.size(-1)).bool(), float('-inf'))
        qk  = query @ key.transpose(-2, -1) / math.sqrt(query.size(-1)) # Compute the dot product of the query and key, and scale it
        qk  = qk + attn_mask # Add the attention mask to the scaled dot product
        attn_weight  = torch.softmax(qk, dim=-1) # Apply softmax to the result
        attn_weight  = torch.dropout(attn_weight, dropout_p=0.3572849639321903, training=self.training) # Apply dropout to the softmax output
        output  = attn_weight @ value # Compute the dot product of the dropout output and the value
        return output
