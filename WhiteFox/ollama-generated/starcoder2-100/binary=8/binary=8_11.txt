
class Model(torch.nn.Module):
    def __init__(self, n):
        super().__init__()
        self.conv  = torch.nn.Conv2d(3, 8, 1)
        self.other = torch.tensor([0] * 5).float()
 
    def forward(self, x1):
        v1  = self.conv(x1)
        v2  = v1 + self.other 
        return v2


# Initializing the model
m  = Model(n=3)

# Inputs to the model
