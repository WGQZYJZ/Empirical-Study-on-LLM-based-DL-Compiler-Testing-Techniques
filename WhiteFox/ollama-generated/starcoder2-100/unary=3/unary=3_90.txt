class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv  = torch.nn.Conv2d(3,8,1)
        self.relu = torch.nn.ReLU()
 
    def forward(self, x1):
        v0  = torch.randn(16,32,4,5)
        v7_ = self.conv(v0) 
        v9  = v7_.sum(dim=0).sqrt() 
        v10 = self.relu(v9)
        v8  = v10 / v7_.var().sqrt() 
        return v8
