
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.fc = torch.nn.Linear(784, 5)
 
    def forward(self, x1):
        v0  = self._reshape(x1)
        v2  = self.fc(v0)
        v3  = torch.sigmoid(v2) 
        return v3
    
    def _reshape(self, tensor):
        return tensor.view(-1, 784)

# Initializing the model
m = Model()

 # Inputs to the model
x1  = torch.randn(64, 784) 
 __output__  = m(x1)
