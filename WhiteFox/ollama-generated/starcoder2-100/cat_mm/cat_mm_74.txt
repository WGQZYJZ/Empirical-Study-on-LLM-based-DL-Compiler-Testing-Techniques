
```
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
 
    def forward(self, x1, x2):
        v1  = torch.mm(x1, x2)
        v2  = torch.cat([v1] * 3, dim=0)
 
        return v2
```

 # Initializing the model and generating input tensors for the new model:
```
m  = Model()
 
x1  = torch.randn(8, 4)
x2  = torch.randn(4, 9)
 
__output__  = m(x1, x2)
