
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv = torch.nn.Conv2d(3, 8, 1)
 
    def forward(self, x1):
        v1 = self.conv(x1)
        v2 = v1 - other_tensor  # Where 'other' is a tensor of the same shape as the output of the convolution and 'other_tensor' can be any tensor in this shape.
        return v2

# Initializing the model
m = Model()


# Inputs to the model
x1 = torch.randn(1, 3, 64, 64)
other_tensor  = torch.randn(8, 1)
__output__  = m(x1, other_tensor)
