class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear  = torch.nn.Linear(1024, 5)
 
    def forward(self, x1):
        v1  = self.linear(x1)
        v2  = torch.clamp_min(v1, -3) # clamps a tensor to its minimum value (-3 in this case)
        v3  = torch.clamp_max(v2, 0) # clamps a tensor to its maximum value (0 in this case)
