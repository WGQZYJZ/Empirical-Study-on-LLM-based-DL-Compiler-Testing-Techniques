
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.query = torch.randn(32, 1000)
        self.key = torch.randn(32, 1000)
        self.value = torch.randn(32, 800, 4096)
 
    def forward(self):
        vq_qk = torch.matmul(self.query, self.key.transpose(-2,-1)) # Compute the dot product of the query and the key
        vs_qk = vq_qk / 3e-5 # Scale the dot product by a constant 3x10^(-5)
        vsoft_vq_qk = vs_qk.softmax(dim=-1) # Apply softmax to the scaled dot product
        vo_vq_qk = torch.nn.functional.dropout(vsoft_vq_qk, p=0.2) # Apply dropout to the softmax output 
        vqo_vo_vq_qk =  self.value * vo_vq_qk.unsqueeze(-1).repeat(1, 1, 4096) # Compute the dot product of the dropout output and a value
        return torch.nn.functional.gelu(vqo_vo_vq_qk)
