
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv = torch.nn.Conv2d(3, 8, 1, stride=1, padding=1)
 
    def forward(self, x1):
        v1  = self.conv(x1) # Apply a pointwise convolution to the input tensor
        return v1

# Initializing the model
m  = Model()


# Inputs to the model
x2  = torch.randn(3, 48, 480//8, 640//8)
__output__  = m(x2)
