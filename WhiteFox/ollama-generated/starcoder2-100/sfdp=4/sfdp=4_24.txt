
class ScaledDotProductAttention(torch.nn.Module):
    def __init__(self):
        super().__init__()
 
    def forward(self, query, key, value): 
        att = torch.matmul(query / math.sqrt(query.size(-1)), key.transpose(-2, -1)) # Dot-product attention. 
        mask_value  = float('-inf') * torch.ones(att.shape).to(query)
        mask_value[:, :, 0] = 0 # Mask out the first row of mask_value to prevent attention to self-attention.
        att += mask_value # Apply the mask. 
        att = torch.softmax(att, dim=-1)  # Softmax over the last axis (i.e., dimension -1). 
        output = torch.matmul(att, value)  # Take dot product of attention weights and value. 
        return output, att
