class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
 
    def forward(self, x1, x2):
        scaled_dot_product  = torch.matmul(x1, x2.transpose(-2, -1)) / inv_scale 
        attention_weights  = scaled_dot_product.softmax(dim=-1)
        output  = attention_weights.matmul(x2)
        return output
