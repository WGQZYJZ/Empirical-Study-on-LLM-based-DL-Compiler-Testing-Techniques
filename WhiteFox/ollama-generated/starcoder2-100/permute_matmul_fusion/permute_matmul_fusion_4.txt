
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()

    def forward(self, x1, x2):

        if torch.__version__ == 'v0':
            t1  = x1.permute(0, 2, 1)  # or t1  = torch.transpose(x1, -1, -2)
        else:                         # or t1  = torch.transpose(x1, 1, 3)
            t1  = x1                   

        if torch.__version__ == 'v0':
            t2  = x2.permute(0, 2, 1)  # or t2  = torch.transpose(x2, -1, -2)
        else:                         # or t2  = torch.transpose(x2, 1, 3)
            t2  = x2

        # The following two lines use the permuted inputs and the permuted tensors for the batch matrix multiplication operation
        v1 = torch.bmm(t1, t2)

        return v1


# Initializing the model
m  = Model()


# Inputs to the model:
x1  = torch.randn(3, 4, 5) # or x1  = torch.randn(1080, 768, 256) in v0
x2  = torch.randn(3, 5, 4) # or x2  = torch.randn(256, 768, 1080) in v0


# Initializing the model
m  = Model()

# Inputs to the model:
__output__  = m(x1, x2)
