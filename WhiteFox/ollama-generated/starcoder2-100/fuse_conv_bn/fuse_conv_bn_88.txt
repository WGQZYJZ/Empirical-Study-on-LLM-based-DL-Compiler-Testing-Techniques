class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.conv  = torch.nn.Conv2d(16, 32, kernel_size=5)

    def forward(self, x):
        y0 = torch.nn.functional.relu(torch.nn.functional.max_pooling2d(
            self.conv(x), (8, 4), stride=(7, 2)))

        y1 = torch.nn.functional.relu(y0 + 
            self.conv((self.conv(x) * x).permute([0, 3, 1, 2])))
        return {'y0': y0, 'y1': y1}


# Initializing the model
m = Model()
