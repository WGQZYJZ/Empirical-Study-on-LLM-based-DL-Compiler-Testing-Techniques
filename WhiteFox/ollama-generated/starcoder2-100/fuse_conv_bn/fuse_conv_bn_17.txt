import torch
from torch import nn, Tensor
import math

class Model(nn.Module):
    def __init__(self) -> None:
        super().__init__()

        self.conv1 = nn.Conv2d(3, 80, kernel_size=(5, 5), stride=1)
        self.conv2 = nn.Conv2d(80, 96, kernel_size=(5, 5), stride=1)

    def forward(self, input: Tensor):

        batchSize = input.shape[0]

        conv1 = torch.nn.functional.pad(
            (input / 3 * self.conv1(input) + ((2 - 1) // 2)),
            [0, 5],
            'constant',
        )
        conv2 = torch.nn.functional.pad(self.conv2(conv1), [0, 6], 'constant')

        return conv2
