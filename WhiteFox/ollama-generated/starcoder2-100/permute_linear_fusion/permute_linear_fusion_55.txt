class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear = torch.nn.Linear(2, 1)

    def forward(self, x1):
        v1 = x1.permute(0, 3, 2, 4) # Permute the input tensor
        v2 = torch.nn.functional.linear(v1[:, :, ::-1], self.linear.weight) 
        return v2


# Initializing the model
m = Model()
