
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear  = torch.nn.Linear(2, 1)

    def forward(self, x1):
        v1  = x1.permute(0, 2, 1) # Permute the input tensor 
        v2  = torch.nn.functional.linear(v1, self.linear.weight, self.linear.bias) # Apply linear transformation to the permuted tensor.
        return v2

# Initializing the model
m = Model()


# Inputs to the model
x1 = torch.randn(4, 50, 784).permute((1, 0, 2)) # Input tensor with shape [N, L, C]
__output__  = m(x1)
