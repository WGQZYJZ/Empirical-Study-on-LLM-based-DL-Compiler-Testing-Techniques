
class Model(torch.nn.Module):
    def __init__(self, num1=2, num2=3):
        super().__init__()

        # Initializing linear module: 1st input_size and output_size is defined by num1 while the number of bias
        # elements is fixed at one for simplicity in model creation.
        self.linear = torch.nn.Linear(num1, num2, bias=False)

    def forward(self, x):
        # Concatenating the input tensor with itself 3 times to get a concatenated tensor.
        v1 = torch.cat([x, x], dim=-1)

        # Reshaping the concatenated tensor into a 4D tensor (4-dimensional tensor). The size of each dimension in this
        # tensor is obtained by dividing the size of input_tensor and number_of_concatenations_before_reshape. This way, we get a 32x2x8x16 4d tensor.
        v2 = v1.view(-1, 8*self.linear(torch.randn(num1)).shape[-1])

        # Applying ReLU to the reshaped tensor
        v3 = torch.nn.functional.relu(v2)

        return v3


# Initializing the model with some initial parameters.
num1= 8; num2=9
m = Model(num1, num2); 

# Inputs for the model:
x  = torch.randn(160*num2)

# Running the forward pass of the model. 
output  = m(x).shape[-1] # The shape should be 8x32.

# In the output above, 8 is determined by the number of bias elements in the linear module and num1=5 while there are 9 tensors concatenated into a single tensor along dimension=-1 which is used as input for reshaping to get the final 4D tensor. 32 is determined using the input_size defined in the linear module and bias set to False.
