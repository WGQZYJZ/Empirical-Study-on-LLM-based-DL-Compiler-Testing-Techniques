    def __init__(self):
        super().__init__()

    def forward(self, x1):
        v  = torch.relu(
            torch.nn.functional
                .linear(
                    torch.cat([x1], dim=0), # Concatenate the inputs along a dimension
                        self.linear
                                .weight, self.linear
                                            .bias) # Apply a linear transformation to the concatenated tensor
            .view(-1))
        return v
