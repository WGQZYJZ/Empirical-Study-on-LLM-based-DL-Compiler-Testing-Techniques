
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.linear = torch.nn.Linear(32768, 50)
 
    def forward(self, x1):
        v1 = self.linear(x1) # Apply the linear transformation to the input tensor
        v2 = v1 + torch.ones_like(v1)
        return v2


# Initializing the model
m = Model()
__output__  = m(torch.randn(3, 4))
