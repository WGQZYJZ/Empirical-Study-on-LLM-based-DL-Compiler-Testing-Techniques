
class Model(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.convtranspose = torch.nn.ConvTranspose2d(3, 8, 1)

    def forward(self, x1):
        v1  = self.convtranspose(x1)
        v2  = torch.tanh(v1)
        return v2

# Initializing the model
m = Model()

 # Inputs to the model
input_tensor = torch.randn(8,3,4,5)

## Running the code:
__output__  = m(input_tensor)
